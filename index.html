/** * SISTEMA DE CÁLCULO REAJUSTADO: * - Taxa Base Fixa: € 50,00 * - Preço por Km (Ida e Volta): € 0,25 / km * - Preço por Kg: € 0,25 / kg * Resultado: València -> Madrid (600 kg) = € 550,00 */ function calculateQuote(event) { if (event) event.preventDefault(); const origin = document.getElementById('origin').value.trim(); const destination = document.getElementById('destination').value.trim(); const weight = parseFloat(document.getElementById('weight').value) || 0; // Distância de ida estimada a partir de València let distanciaIdaKm = 40; const destLower = destination.toLowerCase(); const origLower = origin.toLowerCase(); if (destLower.includes('madrid') || origLower.includes('madrid')) { distanciaIdaKm = 350; } else if (destLower.includes('castellon') || destLower.includes('castellón')) { distanciaIdaKm = 75; } else if (destLower.includes('barcelona')) { distanciaIdaKm = 350; } else if (destLower.includes('alicante')) { distanciaIdaKm = 170; } else if (destLower.includes('valencia') || destLower.includes('valència')) { distanciaIdaKm = 15; } // 1. Distância Total (Ida e Volta Automática) const distanciaTotalKm = distanciaIdaKm * 2; // 2. Parâmetros da Fórmula Ajustada const TAXA_BASE = 50.00; const PRECO_POR_KM = 0.25; // €0.25 por km no percurso total const PRECO_POR_KG = 0.25; // €0.25 por kg // 3. Cálculo Exato const custoDistancia = distanciaTotalKm * PRECO_POR_KM; const custoPeso = weight * PRECO_POR_KG; const valorFinal = TAXA_BASE + custoDistancia + custoPeso; // Exibir Resultado na Tela const resultDiv = document.getElementById('quoteResult'); const priceRange = document.getElementById('priceRange'); const quoteDetails = document.getElementById('quoteDetails'); priceRange.innerText = `€${valorFinal.toFixed(2)}`; quoteDetails.innerHTML = ` Ruta (Ida y Vuelta): ${origin} ➔ ${destination} ➔ ${origin} (${distanciaTotalKm} km totales)
Desglose: Base (€50) + Distancia (€${custoDistancia.toFixed(2)}) + Peso (${weight} kg = €${custoPeso.toFixed(2)})
Nota: Peajes y tiempo de espera excedente se cobran a parte. `; resultDiv.classList.remove('hidden'); }
